# 西安演出集 showHub 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 每日定时从大河票务、大河演出(maitix)、西安儿艺(maitix)、陕西大剧院(snpac) 抓取西安演出，存入 Qoder Sites 数据库并展示在公网页面。

**Architecture:** 纯静态前端（`web/`，无构建步骤）+ 一个 Deno Edge Function（`functions/`，官方 adapter 模式访问 Supabase）。Function 提供三个 action：`list`（分页读）、`meta`（最近同步状态）、`sync`（POST，6 小时限流，抓取→过滤西安→upsert→清理过期）。每日触发由 Qoder 自动化（qoder_cron）调用线上 sync 接口完成。

**Tech Stack:** 原生 HTML/CSS/JS；Deno Edge Function（`npm:@supabase/supabase-js@2.57.4`）；Supabase（Sites 托管数据库）；本地测试用 Node ≥ 22 内置 `node --test`（适配器模块均为无 Deno 依赖的 `.mjs`）。

**Spec:** `docs/superpowers/specs/2026-09-28-showhub-design.md`

## Global Constraints

- 只保存城市为西安的演出；字段：name、poster_url、start_time（原文文本）、start_at（timestamptz，可解析时）、end_time（timestamptz，缺省用开演时间）、price（展示文本）、city、venue、category、status、buy_url。
- 去重键：`(source, source_id)`；source 取值 `dahepiao` / `maitix-dhjc` / `maitix-xaetys` / `snpac`。
- 过期清理：`end_time` 早于 now − 7 天的记录删除。
- 对目标站礼貌抓取：翻页间隔 600ms，dahepiao 最多 6 页，maitix 按接口返回的 totalPage 翻页，snpac 单次请求。
- 展示数据为公开聚合信息，匿名可读；sync 写入经 6 小时限流的同源 POST。
- 适配器互相隔离：单个来源失败记入 sync_log（status=error），不影响其他来源。
- Function 目录内只放运行时代码；测试、fixtures、本地服务器一律放 `dev/`。
- 数据库迁移只用受支持 SQL 子集（uuid/text/integer/timestamptz、CREATE TABLE/CREATE INDEX、无 DEFAULT/CHECK/外键），写入与去重逻辑在 Function 内完成。

```js
// 时间统一按演出地（西安，UTC+8）解析；无法解析返回 null，字段可为空。
export function parseShowDate(text) {
  if (typeof text !== 'string') return null;
  const m = text.match(/(\d{4})-(\d{1,2})-(\d{1,2})(?:[^\d]*?(\d{1,2}):(\d{2}))?/);
  if (!m) return null;
  const pad = (n) => String(n).padStart(2, '0');
  const iso = `${m[1]}-${pad(m[2])}-${pad(m[3])}T${pad(m[4] ?? '0')}:${pad(m[5] ?? '0')}:00+08:00`;
  return Number.isNaN(Date.parse(iso)) ? null : iso;
}

export function toIsoFromMs(ms) {
  return new Date(ms).toISOString();
}

export function msToCSText(ms) {
  // 毫秒时间戳 → 北京墙钟时间文本（用于 start_time 原文列）
  return new Date(ms + 8 * 3600000).toISOString().slice(0, 16).replace('T', ' ');
}

export function formatPriceRange(min, max) {
  const lo = Number(min), hi = Number(max);
  if (!Number.isFinite(lo) || !Number.isFinite(hi) || (lo === 0 && hi === 0)) return '免费/暂无价格';
  if (lo === hi) return `${lo}元`;
  return `${lo}-${hi}元`;
}

export function isXian(city) {
  return typeof city === 'string' && city.includes('西安');
}
```

```js
import { parseShowDate } from './normalize.mjs';

const UA = 'Mozilla/5.0 (iPhone; CPU iPhone OS 16_0 like Mac OS X) AppleWebKit/605.1.15';
const LIST_URL = 'https://m.dahepiao.com/search_list?fenlei=2';

const defaultSleep = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

// 列表页为服务端渲染 HTML；结构：.ycList 卡片内 .s_left img(海报)/.l1 名称/.l2 时间/.l3 场馆/.l4 状态/.l5 价格
export function parseDahepiaoHtml(html) {
  const records = [];
  const blocks = html.split('<div class="ycList list-grid flex">').slice(1);
  for (const block of blocks) {
    const href = block.match(/href="([^"]+)"\s+class="s_left"/);
    const img = block.match(/<img src="([^"]+)"/);
    const name = block.match(/class="l1 line1">\s*([^<]+?)\s*</);
    const time = block.match(/class="l2 line1">\s*([^<]+?)\s*</);
    const venue = block.match(/class="l3 line1">\s*([^<]+?)\s*</);
    const status = block.match(/class="l4">[\s\S]*?<span>\s*([^<]+?)\s*<\/span>/);
    const price = block.match(/class="l5">([\s\S]*?)<\/div>/);
    const id = href && href[1].match(/(\d+)\.html/);
    if (!href || !name || !id) continue;
    const startAt = parseShowDate(time ? time[1] : '');
    records.push({
      source: 'dahepiao',
      source_id: id[1],
      name: name[1].trim(),
      poster_url: img ? img[1].trim() : '',
      start_time: time ? time[1].trim() : '',
      start_at: startAt,
      end_time: startAt,
      price: price ? price[1].replace(/<[^>]+>/g, '').replace(/\s+/g, '').trim() : '',
      city: '西安',
      venue: venue ? venue[1].trim() : '',
      category: '演出',
      status: status ? status[1].trim() : '售票中',
      buy_url: href[1].trim(),
    });
  }
  return records;
}

export async function fetchDahepiao({ transport = globalThis.fetch, sleep = defaultSleep, maxPages = 6 } = {}) {
  const records = [];
  for (let page = 1; page <= maxPages; page += 1) {
    const url = `${LIST_URL}&title=${encodeURIComponent('西安')}&page=${page}`;
    const res = await transport(url, { headers: { 'User-Agent': UA } });
    if (!res.ok) throw new Error(`dahepiao_http_${res.status}`);
    const items = parseDahepiaoHtml(await res.text());
    if (items.length === 0) break;
    records.push(...items);
    if (items.length < 10) break;
    await sleep(600);
  }
  return records;
}
```

```js
import { formatPriceRange, isXian, toIsoFromMs, msToCSText } from './normalize.mjs';

const UA = 'Mozilla/5.0 (iPhone; CPU iPhone OS 16_0 like Mac OS X) AppleWebKit/605.1.15';
const CITY_ID = '610100'; // 西安市

// maitix 平台同 API 不同租户，仅靠 Referer 区分
export const MAITIX_TENANTS = [
  { source: 'maitix-dhjc', referer: 'https://dhjc.maitix.com/', detailBase: 'https://dhjc.maitix.com/m/#/allEvents/detail?projectId=' },
  { source: 'maitix-xaetys', referer: 'https://xaetys.maitix.com/', detailBase: 'https://xaetys.maitix.com/m/#/allEvents/detail?projectId=' },
];

const defaultSleep = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

export function mapMaitixProject(item, tenant) {
  const token = item.projectToken == null ? '' : String(item.projectToken);
  const startAt = typeof item.startTime === 'number' ? toIsoFromMs(item.startTime) : null;
  const endAt = typeof item.endTime === 'number' ? toIsoFromMs(item.endTime) : null;
  return {
    source: tenant.source,
    source_id: token,
    name: String(item.projectName ?? '').trim(),
    poster_url: String(item.imgUrl ?? ''),
    start_time: typeof item.startTime === 'number' ? msToCSText(item.startTime) : '',
    start_at: startAt,
    end_time: endAt ?? startAt,
    price: formatPriceRange(item.minPrice, item.maxPrice),
    city: String(item.cityName ?? ''),
    venue: String(item.siteName ?? ''),
    category: String(item.projectTypeName ?? '演出'),
    status: item.sellOut ? '已售罄' : '售票中',
    buy_url: token ? tenant.detailBase + token : '',
  };
}

export async function fetchMaitixTenant(tenant, { transport = globalThis.fetch, sleep = defaultSleep } = {}) {
```

- [ ] **Step 1: 实现内存版客户端** `dev/fake-supabase.mjs`

```js
// 仅覆盖本项目用到的 Supabase SDK 方法子集；供本地测试与预览，绝不打包进 functions/。
export function fakeSupabase() {
  const tables = new Map();

  const table = (name) => {
    if (!tables.has(name)) tables.set(name, []);
    return tables.get(name);
  };

  const project = (cols, rows) => {
    if (!cols || cols === '*') return rows.map((r) => ({ ...r }));
    const keys = cols.split(',').map((c) => c.trim());
    return rows.map((r) => Object.fromEntries(keys.map((k) => [k, r[k] ?? null])));
  };

  const builder = (name, rows) => {
    const state = { rows: [...rows], mutated: false };
    const chain = {
      select: (cols = '*') => {
        state.cols = cols;
        // .delete().select(...) 之后才返回行；select 起点也返回行
        state.returning = true;
        return chain;
      },
      eq: (col, val) => { state.rows = state.rows.filter((r) => r[col] === val); return chain; },
      in: (col, vals) => { state.rows = state.rows.filter((r) => vals.includes(r[col])); return chain; },
      lt: (col, val) => { state.rows = state.rows.filter((r) => r[col] != null && r[col] < val); return chain; },
      gt: (col, val) => { state.rows = state.rows.filter((r) => r[col] != null && r[col] > val); return chain; },
      order: (col, { ascending = true, nullsFirst = false } = {}) => {
        state.order = { col, ascending, nullsFirst };
        return chain;
      },
      range: (from, to) => { state.range = [from, to]; return chain; },
      limit: (n) => { state.range = [0, n - 1]; return chain; },
      maybeSingle: () => finishOne(),
      single: () => finishOne(),
      insert: (values) => apply((rows) => { rows.push(...clone(values)); return { data: clone(values), error: null }; }),
      upsert: (values, { onConflict } = {}) => apply((rows) => {
        const conflictCols = onConflict ? onConflict.split(',').map((c) => c.trim()) : ['id'];
        const inserted = [];
        for (const v of clone(values)) {
          const idx = rows.findIndex((r) => conflictCols.every((c) => r[c] === v[c]));
          if (idx >= 0) rows[idx] = { ...rows[idx], ...v };
          else rows.push(v);
          inserted.push(v);
        }
        return { data: inserted, error: null };
      }),
      update: (fields) => apply((rows) => {
        const updated = [];
        for (const r of state.rows) {
          if (rows.includes(r)) { Object.assign(r, clone([fields])[0]); updated.push({ ...r }); }
        }
        return { data: updated, error: null };
      }),
      delete: () => {
        state.deleting = true;
        return chain;
      },
    };

    function finishOne() {
      const out = finish();
      return { data: out.data && out.data.length ? out.data[0] : null, error: null };
    }

    function finish() {
      let rows = [...state.rows];
      if (state.order) {
        const { col, ascending, nullsFirst } = state.order;
        const nulls = rows.filter((r) => r[col] == null);
        const vals = rows.filter((r) => r[col] != null).sort((a, b) => {
          const cmp = a[col] < b[col] ? -1 : a[col] > b[col] ? 1 : 0;
          return ascending ? cmp : -cmp;
        });
        rows = nullsFirst ? [...nulls, ...vals] : [...vals, ...nulls];
      }
      if (state.range) rows = rows.slice(state.range[0], state.range[1] + 1);
      const cols = state.cols;
      if (state.deleting) {
        const delRows = state.rows;
        const t = table(name);
        for (const r of delRows) {
          const i = t.indexOf(r);
          if (i >= 0) t.splice(i, 1);
        }
        return { data: project(cols, delRows), error: null };
      }
      return { data: project(cols, rows), error: null };
    }

    const apply = (mutate) => {
      const result = mutate(table(name));
      return result;
    };

    // 所有查询在 await 时结算：对象带 then
    const then = (resolve, reject) => {
      try { resolve(finish()); } catch (e) { reject(e); }
    };
    return new Proxy(chain, {
      get(target, prop, receiver) {
        if (prop === 'then' && !state.mutated) return then;
        const value = Reflect.get(target, prop, receiver);
        return typeof value === 'function' ? value.bind(receiver) : value;
      },
    });
  };

  const clone = (v) => (Array.isArray(v) ? v.map((x) => ({ ...x })) : { ...v });

  return {
    from: (name) => builder(name, table(name)),
    _tables: tables,
  };
}
```

注意：`delete()` 后 `.select(cols)` 需要生效（`state.cols` 被设置且 `state.deleting` 已置位），链式调用 `.delete().lt(...).select('id')` 会先 delete 再 lt 再 select——实现上 `delete()` 只置标记，`lt` 继续过滤，`select` 设列。上表实现满足该顺序。若测试暴露链式顺序问题，以真实 SDK 语义为准修正 fake，而不是改业务代码适配 fake。

```js
import { fetchDahepiao } from './dahepiao.mjs';
import { MAITIX_TENANTS, fetchMaitixTenant } from './maitix.mjs';
import { fetchSnpac } from './snpac.mjs';

const RETENTION_DAYS = 7;

const defaultSleep = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

export const ADAPTERS = [
  { source: 'dahepiao', run: (ctx) => fetchDahepiao(ctx) },
  ...MAITIX_TENANTS.map((tenant) => ({
    source: tenant.source,
    run: (ctx) => fetchMaitixTenant(tenant, ctx),
  })),
  { source: 'snpac', run: (ctx) => fetchSnpac(ctx) },
];

export async function runSync({ supabase, transport = globalThis.fetch, sleep = defaultSleep, now = () => new Date() } = {}) {
  const ctx = { transport, sleep };
  const results = [];
  for (const adapter of ADAPTERS) {
    const row = {
      id: crypto.randomUUID(),
      started_at: now().toISOString(),
      finished_at: null,
      source: adapter.source,
      status: 'error',
      fetched: 0,
      inserted: 0,
      updated: 0,
      deleted: 0,
      error: '',
    };
    try {
      const records = await adapter.run(ctx);
      row.fetched = records.length;
      const counts = await upsertShows(supabase, records, now());
      row.inserted = counts.inserted;
      row.updated = counts.updated;
      row.status = 'success';
    } catch (e) {
      row.error = String((e && e.message) || e).slice(0, 300);
    }
    row.finished_at = now().toISOString();
    const { error } = await supabase.from('sync_log').insert(row);
    if (error) throw new Error('sync_log_write_failed');
    results.push(row);
  }
  const deleted = await cleanupExpired(supabase, now());
  const cleanupRow = {
    id: crypto.randomUUID(),
    started_at: now().toISOString(),
    finished_at: now().toISOString(),
    source: 'cleanup',
    status: 'success',
    fetched: 0,
    inserted: 0,
    updated: 0,
    deleted,
    error: '',
  };
  await supabase.from('sync_log').insert(cleanupRow);
  return { results, deleted };
}

async function upsertShows(supabase, records, now) {
  if (records.length === 0) return { inserted: 0, updated: 0 };
  const source = records[0].source;
  const ids = records.map((r) => r.source_id);
  const { data: existing, error } = await supabase
    .from('shows')
    .select('source_id')
    .eq('source', source)
    .in('source_id', ids);
  if (error || !Array.isArray(existing)) throw new Error('shows_read_failed');
  const existingIds = new Set((existing ?? []).map((r) => r.source_id));
  const rows = records.map((r) => ({ ...r, id: crypto.randomUUID(), updated_at: now.toISOString() }));
  const { error: upsertError } = await supabase
    .from('shows')
    .upsert(rows, { onConflict: 'source,source_id' });
  if (upsertError) throw new Error('shows_write_failed');
  const inserted = rows.filter((r) => !existingIds.has(r.source_id)).length;
  return { inserted, updated: rows.length - inserted };
}

async function cleanupExpired(supabase, now) {
```

```js
import { runSync } from './sync/sync.mjs';

const SYNC_INTERVAL_MS = 6 * 3600000;

const json = (body, status = 200, headers = {}) =>
  Response.json(body, { status, headers: { 'cache-control': 'no-store', ...headers } });

export async function handler({ request, supabase }) {
  const params = new URL(request.url).searchParams;
  const action = params.get('action');
  if (action === 'list') return handleList(request, params, supabase);
  if (action === 'meta') return handleMeta(supabase);
  if (action === 'sync') return handleSync(request, supabase);
  return json({ error: 'not_found' }, 404);
}

async function handleList(request, params, supabase) {
  if (request.method !== 'GET') return json({ error: 'method_not_allowed' }, 405, { allow: 'GET' });
  const rawOffset = params.get('offset') ?? '0';
  if (!/^(0|[1-9][0-9]{0,5})$/.test(rawOffset)) return json({ error: 'invalid_offset' }, 400);
  const offset = Number(rawOffset);
  const pageSize = 200;
  try {
    const { data, error } = await supabase.from('shows')
      .select('source,source_id,name,poster_url,start_time,start_at,end_time,price,city,venue,category,status,buy_url')
      .order('start_at', { ascending: true, nullsFirst: false })
      .order('name', { ascending: true })
      .range(offset, offset + pageSize);
    if (error || !Array.isArray(data)) return json({ error: 'database_request_failed' }, 503);
    const hasMore = data.length > pageSize;
    return json({ items: data.slice(0, pageSize), hasMore, nextOffset: hasMore ? offset + pageSize : null });
  } catch {
    return json({ error: 'database_request_failed' }, 503);
  }
}

async function handleMeta(supabase) {
  try {
    const { data, error } = await supabase.from('sync_log')
      .select('source,status,finished_at,fetched,inserted,updated,deleted')
      .order('finished_at', { ascending: false })
      .range(0, 29);
    if (error || !Array.isArray(data)) return json({ error: 'database_request_failed' }, 503);
    const sources = [];
    const seen = new Set();
    let lastSuccessAt = null;
    for (const row of data) {
      if (row.source === 'cleanup' && !seen.has('cleanup')) { seen.add('cleanup'); continue; }
      if (!seen.has(row.source)) {
        seen.add(row.source);
        sources.push({
          source: row.source,
          status: row.status,
          finished_at: row.finished_at,
          fetched: row.fetched,
          inserted: row.inserted,
          updated: row.updated,
          deleted: row.deleted,
        });
        if (row.status === 'success' && row.source !== 'cleanup' && (!lastSuccessAt || row.finished_at > lastSuccessAt)) {
          lastSuccessAt = row.finished_at;
        }
      }
    }
    return json({ sources, lastSuccessAt });
  } catch {
    return json({ error: 'database_request_failed' }, 503);
  }
}

async function handleSync(request, supabase) {
  if (request.method !== 'POST') return json({ error: 'method_not_allowed' }, 405, { allow: 'POST' });
  try {
    const { data, error } = await supabase.from('sync_log')
      .select('finished_at')
      .eq('status', 'success')
      .neq('source', 'cleanup')
      .order('finished_at', { ascending: false })
      .range(0, 0);
    if (error || !Array.isArray(data)) return json({ error: 'database_request_failed' }, 503);
    const last = data[0];
    if (last && last.finished_at && Date.now() - Date.parse(last.finished_at) < SYNC_INTERVAL_MS) {
      return json({ error: 'sync_rate_limited' }, 429);
    }
    const summary = await runSync({ supabase });
    return json({ ok: true, ...summary });
  } catch {
    return json({ error: 'sync_failed' }, 503);
  }
}
```

```js
  '.html': 'text/html; charset=utf-8',
  '.js': 'text/javascript; charset=utf-8',
  '.css': 'text/css; charset=utf-8',
  '.svg': 'image/svg+xml',
  '.png': 'image/png',
  '.jpg': 'image/jpeg',

  try {
    const url = new URL(req.url, `http://127.0.0.1:${port}`);
    if (url.pathname.startsWith('/functions/v1/app')) {
      const body = await new Promise((resolve) => {
        const chunks = [];
        req.on('data', (c) => chunks.push(c));
        req.on('end', () => resolve(Buffer.concat(chunks).toString()));
      });
      const request = new Request(url.href, { method: req.method, headers: req.headers, body: ['GET', 'HEAD'].includes(req.method) ? undefined : body });
      const response = await handler({ request, supabase });
      res.writeHead(response.status, Object.fromEntries(response.headers));
      res.end(await response.text());
      return;
    }
    let path = url.pathname === '/' ? '/index.html' : url.pathname;
    const file = normalize(join(root, path));
    if (!file.startsWith(root)) { res.writeHead(403); res.end(); return; }
    const content = await readFile(file);
    res.writeHead(200, { 'content-type': mime[extname(file)] ?? 'application/octet-stream' });
    res.end(content);
  } catch {
    res.writeHead(404, { 'content-type': 'text/plain; charset=utf-8' });
    res.end('Not Found');
  }
}).listen(port, '127.0.0.1', () => {
```
